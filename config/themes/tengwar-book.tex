\newcommand{\ThemeSetup}{%
  \KOMAoptions{headinclude=false,footinclude=false}
  \areaset[9mm]{145mm}{205mm}
  \renewcommand*{\raggedchapter}{\centering}
  \renewcommand*{\chapterlinesformat}[3]{%
    \Ifstr{##1}{chapter}{%
      \parbox{\linewidth}{\centering {##2}\par\vspace{0.75\baselineskip}##3\par}%
    }{##2##3}}
  \RedeclareSectionCommand[beforeskip=1.0cm,afterskip=1.1cm]{chapter}
  \clearpairofpagestyles
  \automark[section]{chapter}
  \lefoot{\small\rmfamily\pagemark\enspace\headmark}
  \rofoot{\small\rmfamily\headmark\enspace\pagemark}
  \renewcommand*{\chapterpagestyle}{plain.scrheadings}
  \setkomafont{pageheadfoot}{\rmfamily\color{DeepInk}}
  \setkomafont{subsection}{\rmfamily\itshape\color{DeepInk}}
  \setkomafont{captionlabel}{\rmfamily\bfseries\color{DeepInk}}
}
